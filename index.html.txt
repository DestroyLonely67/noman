<script>
    // --------------------------------------------------------------
    // RAW DISCORD WS IMPLEMENTATION - LIGHTWEIGHT, NO LIBRARIES
    // Connect via Gateway v10 with minimal OP codes, heartbeat, and presence.
    // --------------------------------------------------------------
    (function() {
        // DOM elements
        const tokenInput = document.getElementById('botToken');
        const startBtn = document.getElementById('startBtn');
        const stopBtn = document.getElementById('stopBtn');
        const logArea = document.getElementById('logArea');
        const statusLed = document.getElementById('statusLed');

        // Bot state
        let ws = null;              // WebSocket connection
        let heartbeatInterval = null;
        let lastSeq = null;        // sequence for resume (we ignore resume but keep)
        let sessionId = null;
        let heartbeatAck = true;
        let isBotActive = false;
        let currentToken = null;
        let reconnectAttempt = 0;
        let intentionalClose = false;

        // Helper: add log message
        function addLog(message, type = 'info') {
            const logDiv = document.createElement('div');
            logDiv.className = `log-line ${type}`;
            const timestamp = new Date().toLocaleTimeString('en-US', { hour12: false });
            logDiv.innerHTML = `<span class="timestamp">[${timestamp}]</span> ${escapeHtml(message)}`;
            logArea.appendChild(logDiv);
            logDiv.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
            // keep log tidy but limit 500 lines (performance)
            while(logArea.children.length > 380) {
                logArea.removeChild(logArea.firstChild);
            }
        }

        function escapeHtml(str) {
            if (!str) return '';
            return str.replace(/[&<>]/g, function(m) {
                if (m === '&') return '&amp;';
                if (m === '<') return '&lt;';
                if (m === '>') return '&gt;';
                return m;
            }).replace(/[\uD800-\uDBFF][\uDC00-\uDFFF]/g, function(c) {
                return c;
            });
        }

        function updateLed(online) {
            if (online) {
                statusLed.classList.remove('offline');
                statusLed.style.background = '#2bff9c';
                statusLed.style.boxShadow = '0 0 10px #2bff9c';
            } else {
                statusLed.classList.add('offline');
                statusLed.style.background = '#ff3b6f';
                statusLed.style.boxShadow = '0 0 5px #ff3b6f';
            }
        }

        // stop all bot connections & clean intervals
        function fullCleanup(resetTokenState = false) {
            if (heartbeatInterval) {
                clearInterval(heartbeatInterval);
                heartbeatInterval = null;
            }
            if (ws && (ws.readyState === WebSocket.OPEN || ws.readyState === WebSocket.CONNECTING)) {
                try {
                    ws.close(1000, 'Manual stop');
                } catch(e) { /* ignore */ }
            }
            ws = null;
            heartbeatAck = true;
            isBotActive = false;
            updateLed(false);
            if (resetTokenState) {
                currentToken = null;
            }
            intentionalClose = false;
        }

        // send gateway payload
        function sendPayload(payload) {
            if (ws && ws.readyState === WebSocket.OPEN) {
                ws.send(JSON.stringify(payload));
                return true;
            }
            return false;
        }

        // identify (resume not implemented fully, fresh IDENTIFY)
        function identifyWithToken(token) {
            const identifyPayload = {
                op: 2,
                d: {
                    token: token,
                    intents: 513,   // GUILDS + GUILD_MESSAGES + MESSAGE_CONTENT (basic but enough for online status)
                    properties: {
                        os: process.platform || "browser",
                        browser: "DiscordBotWebTerminal",
                        device: "token_launcher"
                    },
                    presence: {
                        status: "online",
                        afk: false,
                        activities: [{
                            name: "slash commands",
                            type: 0
                        }]
                    }
                }
            };
            sendPayload(identifyPayload);
            addLog("🔐 Sending IDENTIFY payload → gateway", "info");
        }

        // heartbeat sender
        function startHeartbeat(intervalMs) {
            if (heartbeatInterval) clearInterval(heartbeatInterval);
            heartbeatInterval = setInterval(() => {
                if (ws && ws.readyState === WebSocket.OPEN) {
                    if (!heartbeatAck) {
                        addLog("💔 Heartbeat ACK missed. Force closing connection.", "error");
                        fullCleanup();
                        addLog("❌ Connection terminated due to unresponsive gateway.", "error");
                        isBotActive = false;
                        updateLed(false);
                        return;
                    }
                    heartbeatAck = false;
                    sendPayload({ op: 1, d: lastSeq !== null ? lastSeq : null });
                } else {
                    if (heartbeatInterval) clearInterval(heartbeatInterval);
                    heartbeatInterval = null;
                }
            }, intervalMs);
        }

        // core websocket initialization
        function initWebSocket(token) {
            if (ws && ws.readyState === WebSocket.OPEN) {
                addLog("⚠️ There's already an active connection. Stop current bot first.", "error");
                return false;
            }
            intentionalClose = false;
            const gatewayUrl = "wss://gateway.discord.gg/?v=10&encoding=json";
            addLog(`🌐 Connecting to Discord Gateway (v10)...`, "info");
            ws = new WebSocket(gatewayUrl);
            ws.onopen = () => {
                addLog("✅ WebSocket opened. Waiting for hello...", "success");
            };
            ws.onmessage = (event) => {
                try {
                    const data = JSON.parse(event.data);
                    handleGatewayMessage(data, token);
                } catch (err) {
                    addLog(`⚠️ Failed parsing gateway message: ${err.message}`, "error");
                }
            };
            ws.onerror = (err) => {
                addLog(`🔌 WebSocket error: ${err.type || 'connection error'}`, "error");
            };
            ws.onclose = (closeEvent) => {
                const wasActive = isBotActive;
                if (heartbeatInterval) clearInterval(heartbeatInterval);
                heartbeatInterval = null;
                isBotActive = false;
                updateLed(false);
                if (!intentionalClose && wasActive) {
                    addLog(`⚠️ Connection closed (code ${closeEvent.code}) - ${closeEvent.reason || 'no reason'}`, "error");
                    if (reconnectAttempt < 3) {
                        reconnectAttempt++;
                        addLog(`🔄 Auto reconnect attempt ${reconnectAttempt}/3 in 3s...`, "info");
                        setTimeout(() => {
                            if (!intentionalClose && token === currentToken && token) {
                                addLog(`♻️ Reconnecting...`, "info");
                                initWebSocket(token);
                            }
                        }, 3000);
                    } else {
                        addLog(`❌ Max reconnects reached. Please start manually.`, "error");
                        reconnectAttempt = 0;
                    }
                } else {
                    addLog(`🔌 Bot disconnected (intentional)`, "info");
                }
                ws = null;
            };
            return true;
        }

        // main message handler
        function handleGatewayMessage(data, token) {
            const { op, d, t, s } = data;
            if (s) lastSeq = s;
            switch(op) {
                case 10: // HELLO (heartbeat interval)
                    const heartbeatIntervalMs = d.heartbeat_interval;
                    addLog(`💓 Heartbeat interval: ${heartbeatIntervalMs}ms`, "success");
                    startHeartbeat(heartbeatIntervalMs);
                    // after hello, send identify
                    identifyWithToken(token);
                    break;
                case 11: // HEARTBEAT ACK
                    heartbeatAck = true;
                    break;
                case 0: // DISPATCH
                    if (t === 'READY') {
                        sessionId = d.session_id;
                        const user = d.user;
                        addLog(`🎉 BOT ONLINE! Logged in as: ${user.username}#${user.discriminator || '0'} (ID: ${user.id})`, "success");
                        isBotActive = true;
                        updateLed(true);
                        reconnectAttempt = 0;
                        addLog(`✨ Bot is ready — listening to gateway events.`, "success");
                    } else if (t === 'RESUMED') {
                        addLog(`🔄 Session resumed successfully.`, "success");
                        isBotActive = true;
                        updateLed(true);
                    } else if (t === 'GUILD_CREATE') {
                        const guildName = d.name;
                        addLog(`📁 Joined guild / loaded: ${guildName} (${d.id})`, "info");
                    } else {
                        // optional: minimal logging for message create
                        if (t === 'MESSAGE_CREATE' && d.author && d.author.id !== d.application_id) {
                            const author = d.author.username;
                            const content = d.content?.substring(0, 50);
                            if (content) addLog(`💬 ${author}: ${escapeHtml(content)}${d.content?.length > 50 ? '…' : ''}`, "info");
                        }
                    }
                    break;
                case 9: // INVALID SESSION
                    addLog(`🚫 Invalid session (maybe token invalid or login expired).`, "error");
                    fullCleanup();
                    updateLed(false);
                    isBotActive = false;
                    break;
                default:
                    // ignore other OP codes
                    break;
            }
        }

        // --------------------------------------------------------------
        // PUBLIC CONTROLS
        // --------------------------------------------------------------
        function startBot() {
            if (isBotActive) {
                addLog("⚠️ Bot is already running. Stop it first.", "error");
                return;
            }
            let token = tokenInput.value.trim();
            if (!token) {
                addLog("❌ No token provided. Paste your Discord bot token.", "error");
                return;
            }
            // basic token format validation (Discord tokens are alphanumeric + dots + underscores)
            if (!token.match(/^[A-Za-z0-9._-]+$/)) {
                addLog("⚠️ Token contains suspicious characters. Double-check.", "error");
            }
            if (token.length < 50) {
                addLog("⚠️ Token seems too short. Verify if it's valid.", "error");
            }
            currentToken = token;
            // Clear previous connection fully
            fullCleanup(false);
            intentionalClose = false;
            addLog(`🚀 Attempting to start Discord bot with provided token...`, "info");
            addLog(`🔒 Connecting securely via WebSocket.`, "info");
            initWebSocket(token);
        }

        function stopBot() {
            if (!isBotActive && (!ws || ws.readyState === WebSocket.CLOSED)) {
                addLog("ℹ️ No active bot connection to stop.", "info");
                fullCleanup(false);
                updateLed(false);
                return;
            }
            intentionalClose = true;
            addLog(`⏹ Stopping bot... disconnecting from gateway.`, "info");
            fullCleanup(false);
            isBotActive = false;
            updateLed(false);
            addLog(`✅ Bot stopped. You can start again with a token.`, "success");
            reconnectAttempt = 0;
        }

        // attach event listeners
        startBtn.addEventListener('click', startBot);
        stopBtn.addEventListener('click', stopBot);
        // optional: press enter in token field does nothing accidental
        tokenInput.addEventListener('keypress', (e) => {
            if (e.key === 'Enter') {
                e.preventDefault();
                startBot();
            }
        });

        // Initial sync: ensure stopped state
        updateLed(false);
        addLog("🟢 Ready: paste token → click START. Bot will appear online.", "info");
        addLog("📌 Note: Only tokens with 'bot' prefix work. Intents: GUILDS, MESSAGES.", "info");
    })();
</script>
